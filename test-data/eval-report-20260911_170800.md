- 生成的 SQL：```sql
WITH vehicle_weight AS (
  SELECT rm.region_id,
         rm.vehicle_id,
         SUM(rm.total_weight) AS total_weight
  FROM t_route_manifest rm
  GROUP BY rm.region_id, rm.vehicle_id
),
ranked AS (
  SELECT vw.region_id,
         vw.vehicle_id,
         vw.total_weight,
         ROW_NUMBER() OVER (PARTITION BY vw.region_id ORDER BY vw.total_weight DESC) AS rn
  FROM vehicle_weight vw
)
SELECT r.region_name,
       v.plate_number,
       ranked.total_weight AS total_weight_kg
FROM ranked
JOIN t_region r ON r.id = ranked.region_id
JOIN t_vehicle v ON v.id = ranked.vehicle_id
WHERE ranked.rn = 1
ORDER BY ranked.total_weight DESC
LIMIT 1000
```

- 说明：结果不一致：实际 6 行 / 期望 6 行
- 生成的 SQL：```sql
SELECT v.plate_number, r.region_name, COUNT(*) AS alert_count, MAX(a.alert_time) AS last_alert_time FROM t_alert a JOIN t_vehicle v ON a.vehicle_id = v.id LEFT JOIN t_region r ON v.region_id = r.id WHERE a.alert_type LIKE '%超速%' AND a.alert_time >= CURRENT_DATE - INTERVAL '3 days' GROUP BY v.plate_number, r.region_name ORDER BY alert_count DESC, last_alert_time DESC LIMIT 1000
