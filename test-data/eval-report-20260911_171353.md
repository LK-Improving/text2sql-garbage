WITH month_manifest AS (
  SELECT vehicle_id, SUM(total_weight) AS total_weight
  FROM t_route_manifest
  WHERE route_date >= date_trunc('month', CURRENT_DATE)::date
    AND route_date < (date_trunc('month', CURRENT_DATE) + INTERVAL '1 month')::date
  GROUP BY vehicle_id
),
ranked AS (
  SELECT vehicle_id, total_weight, RANK() OVER (ORDER BY total_weight DESC) AS rnk
  FROM month_manifest

top5 AS (
  SELECT vehicle_id, total_weight
  FROM ranked
  WHERE rnk <= 5
),
overspeed AS (
  SELECT DISTINCT vehicle_id
  FROM t_alert

    AND alert_time >= date_trunc('month', CURRENT_DATE)
    AND alert_time < date_trunc('month', CURRENT_DATE) + INTERVAL '1 month'
)
SELECT v.plate_number, r.region_name, t.total_weight
FROM top5 t
JOIN overspeed o ON t.vehicle_id = o.vehicle_id
JOIN t_vehicle v ON t.vehicle_id = v.id
JOIN t_region r ON v.region_id = r.id
ORDER BY t.total_weight DESC
LIMIT 1000
